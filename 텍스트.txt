            function updateCharacterPose() {
                const h0 = params.releaseHeight;
                const standingReach = 1.58;
                const minCrouchReach = 0.85;

                let targetBaseY = 0;
                let targetScaleY = 1.0;
                let targetArmAngle = -0.18;

                if (h0 >= standingReach) {
                    targetBaseY = h0 - standingReach;
                    targetArmAngle = -0.12;
                } else if (h0 >= minCrouchReach) {
                    const ratio = (h0 - minCrouchReach) / (standingReach - minCrouchReach);
                    targetScaleY = 0.72 + 0.28 * ratio;
                    targetArmAngle = -Math.PI * (0.65 - 0.53 * ratio);
                } else {
                    targetScaleY = 0.65;
                    targetArmAngle = -Math.PI * 0.85;
                }

                let charBaseY = 0;
                let charScaleY = 1.0;
                let armTiltAngle = -0.85;

                // 기본 자세 회전 초기화
                playerChar.rotation.set(0, 0, 0);
                headGroup.rotation.set(0, 0, 0);
                leftArm.rotation.set(0, 0, 0);
                rightArm.rotation.set(0, 0, 0);

                // ★ 득점 시 물만난 광어 댄스 세리머니 모션!
                if (state.celebrateTimer > 0) {
                    const ct = state.celebrateTimer;
                    const flopWave = Math.sin(ct * 22);       // 빠른 광어 펄떡임 주기
                    const tailWave = Math.sin(ct * 22 - 1.1); // 꼬리(다리) 시차 파동
                    const bounce = Math.abs(Math.sin(ct * 11)) * 0.16; // 바닥에서 통통 튀어 오름

                    // 코트 바닥에 옆으로 누운 자세 + 좌우/상하 파닥거림
                    playerChar.rotation.x = Math.PI * 0.48 + flopWave * 0.14;
                    playerChar.rotation.z = Math.sin(ct * 11) * 0.18;
                    charBaseY = 0.24 + bounce;
                    charScaleY = 1.0;

                    // 머리(아가미)와 다리(꼬리지느러미)가 번갈아 펄떡이는 광어 웨이브
                    headGroup.rotation.z = flopWave * 0.35;
                    leftLeg.rotation.z = tailWave * 0.65;
                    rightLeg.rotation.z = -tailWave * 0.65;
                    leftShoe.position.x = 0.04 - tailWave * 0.14;
                    rightShoe.position.x = 0.04 + tailWave * 0.14;

                    // 양팔(가슴지느러미) 파닥파닥
                    armTiltAngle = -1.2 + flopWave * 0.45;
                    leftArm.rotation.x = 0.4 + flopWave * 0.5;
                    rightArm.rotation.x = -0.4 - flopWave * 0.5;

                    shooterGroup.position.set(playerPos.x, 0, playerPos.z);
                    shooterGroup.rotation.y = -playerPos.aimYaw;
                    playerChar.position.set(-0.18, charBaseY, 0);
                    playerChar.scale.set(1, charScaleY, 1);
                    armsPivot.rotation.z = armTiltAngle;
                    releaseRing.position.set(0, h0, 0);
                    return;
                }

                if (state.status === 'ready') {
                    charBaseY = playerPos.isMoving ? Math.abs(Math.sin(playerPos.runCycle)) * 0.08 : 0;
                    charScaleY = 1.0;
                    armTiltAngle = -0.85;
                } else if (state.status === 'jumping') {
                    const p = Math.min(1, state.jumpTimer / state.jumpDuration);
                    if (p < 0.3) {
                        const crouchP = p / 0.3;
                        const easeCrouch = Math.sin(crouchP * Math.PI * 0.5);
                        charBaseY = 0;
                        charScaleY = 1.0 - 0.14 * easeCrouch;
                        armTiltAngle = -0.85 - 0.15 * easeCrouch;
                    } else {
                        const jumpP = (p - 0.3) / 0.7;
                        const easeJump = 1 - Math.pow(1 - jumpP, 2);
                        charBaseY = targetBaseY * easeJump;
                        charScaleY = (1.0 - 0.14) + (targetScaleY - (1.0 - 0.14)) * easeJump;
                        armTiltAngle = -1.0 + (targetArmAngle - (-1.0)) * easeJump;
                    }
                } else {
                    const landProgress = Math.min(1, state.t / 0.42);
                    const easeLand = 1 - Math.pow(1 - landProgress, 2);
                    charBaseY = targetBaseY * (1 - easeLand);
                    charScaleY = targetScaleY + (1.0 - targetScaleY) * easeLand;
                    const followThrough = -0.32 * Math.sin(Math.min(1, state.t / 0.25) * Math.PI * 0.5);
                    armTiltAngle = targetArmAngle + followThrough;
                }

                if (state.status === 'ready' && playerPos.isMoving) {
                    const swing = Math.sin(playerPos.runCycle) * 0.55;
                    leftLeg.rotation.z = swing;
                    rightLeg.rotation.z = -swing;
                    leftShoe.position.x = 0.04 - Math.sin(playerPos.runCycle) * 0.12;
                    rightShoe.position.x = 0.04 + Math.sin(playerPos.runCycle) * 0.12;
                } else {
                    leftLeg.rotation.z = 0;
                    rightLeg.rotation.z = 0;
                    leftShoe.position.x = 0.04;
                    rightShoe.position.x = 0.04;
                }

                shooterGroup.position.set(playerPos.x, 0, playerPos.z);
                shooterGroup.rotation.y = -playerPos.aimYaw;

                playerChar.position.set(-0.18, charBaseY, 0);
                playerChar.scale.set(1, charScaleY, 1);
                armsPivot.rotation.z = armTiltAngle;

                releaseRing.position.set(0, h0, 0);
            }

            const trailIndices = [];

            function update3DTrail() {
                const show = toggleTrail.checked && state.trail.length >= 2;
                trailRibbonMesh.visible = show;
                coreLineMesh.visible = show;
                if (!show) return;

                const sliceStart = Math.max(0, state.trail.length - MAX_TRAIL_POINTS);
                const count = state.trail.length - sliceStart;

                const tailColor = new THREE.Color(0xef4444);
                const midColor = new THREE.Color(0xf97316);
                const headColor = new THREE.Color(0xfef08a);

                for (let i = 0; i < MAX_TRAIL_POINTS; i++) {
                    const idx = Math.min(count - 1, i);
                    const p = state.trail[sliceStart + idx];
                    const prevP = state.trail[sliceStart + Math.max(0, idx - 1)];
                    const nextP = state.trail[sliceStart + Math.min(count - 1, idx + 1)];

                    let tx = nextP.x - prevP.x;
                    let ty = nextP.y - prevP.y;
                    let tz = (nextP.z || 0) - (prevP.z || 0);
                    const len = Math.hypot(tx, ty, tz) || 1;
                    tx /= len;
                    ty /= len;
                    tz /= len;

                    const horizDirX = Math.cos(playerPos.aimYaw);
                    const horizDirZ = Math.sin(playerPos.aimYaw);
                    const nx = -ty * horizDirX;
                    const ny = tx * horizDirX + tz * horizDirZ;
                    const nz = -ty * horizDirZ;

                    const ratio = count > 1 ? idx / (count - 1) : 1;
                    const halfWidth = BALL_RADIUS * 0.88 * Math.pow(ratio, 1.35);

                    const vIdx = i * 6;
                    trailPositions[vIdx]     = p.x + nx * halfWidth;
                    trailPositions[vIdx + 1] = p.y + ny * halfWidth;
                    trailPositions[vIdx + 2] = (p.z || 0) + nz * halfWidth;

                    trailPositions[vIdx + 3] = p.x - nx * halfWidth;
                    trailPositions[vIdx + 4] = p.y - ny * halfWidth;
                    trailPositions[vIdx + 5] = (p.z || 0) - nz * halfWidth;

                    const c = new THREE.Color();
                    if (ratio < 0.5) {
                        c.lerpColors(tailColor, midColor, ratio * 2);
                    } else {
                        c.lerpColors(midColor, headColor, (ratio - 0.5) * 2);
                    }
                    c.multiplyScalar(0.25 + 0.75 * ratio);

                    trailColors[vIdx]     = c.r;
                    trailColors[vIdx + 1] = c.g;
                    trailColors[vIdx + 2] = c.b;
                    trailColors[vIdx + 3] = c.r;
                    trailColors[vIdx + 4] = c.g;
                    trailColors[vIdx + 5] = c.b;

                    coreLinePositions[i * 3]     = p.x;
                    coreLinePositions[i * 3 + 1] = p.y;
                    coreLinePositions[i * 3 + 2] = (p.z || 0) + 0.01;
                }

                trailGeo.attributes.position.needsUpdate = true;
                trailGeo.attributes.color.needsUpdate = true;
                trailGeo.setDrawRange(0, (count - 1) * 6);

                coreLineGeo.attributes.position.needsUpdate = true;
                coreLineGeo.setDrawRange(0, count);
            }

            function getTheoreticalMaxHeight() {
                const rad = (params.angleDeg * Math.PI) / 180;
                const vy0 = params.speed * Math.sin(rad);
                return params.releaseHeight + (vy0 * vy0) / (2 * GRAVITY);

            function update3DEnvironmentAndGuides() {
                hoopGroup.position.x = HOOP_WORLD_X;
                rimPointLight.position.x = HOOP_WORLD_X;

                heightPole.scale.set(1, Math.max(0.05, params.releaseHeight), 1);

                aimGuideGroup.visible = toggleAimGuide.checked;
                if (toggleAimGuide.checked) {
                    const rad = (params.angleDeg * Math.PI) / 180;
                    const vx0 = params.speed * Math.cos(rad);
                    const vy0 = params.speed * Math.sin(rad);

                    for (let i = 0; i < aimDots.length; i++) {
                        const t = (i + 1) * 0.045;
                        const ps = vx0 * t;
                        const py = params.releaseHeight + vy0 * t - 0.5 * GRAVITY * t * t;
                        if (py >= BALL_RADIUS && ps <= params.hoopDistance + 2.5) {
                            const worldXZ = getWorldXZFromProgress(ps);
                            aimDots[i].visible = true;
                            aimDots[i].position.set(worldXZ.x, py, worldXZ.z);
                        } else {
                            aimDots[i].visible = false;
                        }
                    }
                }

                const maxH = getTheoreticalMaxHeight();
                overlayApexVal.textContent = maxH.toFixed(2) + ' m';
                valMaxH.textContent = maxH.toFixed(2) + ' m';

                apexMarkerGroup.visible = toggleApexMarker.checked;
                if (toggleApexMarker.checked) {
                    if (state.apexPoint) {
                        apexMarkerGroup.position.set(state.apexPoint.x, state.apexPoint.y, state.apexPoint.z || 0);
                    } else {
                        const rad = (params.angleDeg * Math.PI) / 180;
                        const vx0 = params.speed * Math.cos(rad);
                        const vy0 = params.speed * Math.sin(rad);
                        const tApex = Math.max(0, vy0 / GRAVITY);
                        const apexXZ = getWorldXZFromProgress(vx0 * tApex);
                        apexMarkerGroup.position.set(apexXZ.x, maxH, apexXZ.z);
                    }
                }
            }

            // 매 프레임 공 및 속도 벡터 3D 위치 업데이트
            function sync3DObjects() {
                updateCharacterPose();

                // 준비('ready') 또는 점프 중('jumping')일 때는 공이 선수의 양손 위치를 따라 이동
                if (state.status === 'ready' || state.status === 'jumping') {
                    playerChar.updateMatrixWorld(true);
                    const leftHandWorld = new THREE.Vector3();
                    const rightHandWorld = new THREE.Vector3();
                    leftHand.getWorldPosition(leftHandWorld);
                    rightHand.getWorldPosition(rightHandWorld);
                    state.x = (leftHandWorld.x + rightHandWorld.x) * 0.5 + Math.cos(playerPos.aimYaw) * 0.06;
                    state.y = (leftHandWorld.y + rightHandWorld.y) * 0.5 + 0.04;
                    state.z = (leftHandWorld.z + rightHandWorld.z) * 0.5 + Math.sin(playerPos.aimYaw) * 0.06;
                }

                ballMesh.position.set(state.x, state.y, state.z);
                ballMesh.rotation.y = -playerPos.aimYaw;
                ballMesh.rotation.z = state.spinAngle;

                // 바닥 그림자 위치 및 크기 조절
                floorShadowMesh.position.set(state.x, 0.015, state.z);
                const shadowScale = Math.max(0.35, 1 - state.y * 0.12);
                floorShadowMesh.scale.set(shadowScale, shadowScale, 1);
                floorShadowMesh.material.opacity = Math.max(0.12, 0.48 - state.y * 0.06);

                // 3D 속도 벡터 화살표 갱신 (공이 실제로 발사된 이후에만 표시, 골대 방향 수직 평면 기준)
                const isLaunched = state.status === 'running' || state.status === 'paused' || state.status === 'finished';
                const showVec = toggleVectors.checked && isLaunched;
                arrowVx.visible = showVec && Math.abs(state.vx) > 0.05;
                arrowVy.visible = showVec && Math.abs(state.vy) > 0.05;
                const speed = Math.hypot(state.vx, state.vy);
                arrowV.visible = showVec && speed > 0.05;

                if (showVec) {
                    const origin = new THREE.Vector3(state.x, state.y, state.z);
                    const vecScale = 0.16;
                    const dirX = Math.cos(playerPos.aimYaw);
                    const dirZ = Math.sin(playerPos.aimYaw);

                    if (arrowVx.visible) {
                        arrowVx.position.copy(origin);
                        arrowVx.setDirection(new THREE.Vector3(dirX * Math.sign(state.vx), 0, dirZ * Math.sign(state.vx)).normalize());
                        arrowVx.setLength(Math.max(0.28, Math.abs(state.vx) * vecScale), 0.18, 0.10);
                    }
                    if (arrowVy.visible) {
                        arrowVy.position.copy(origin);
                        arrowVy.setDirection(new THREE.Vector3(0, Math.sign(state.vy), 0));
                        arrowVy.setLength(Math.max(0.28, Math.abs(state.vy) * vecScale), 0.18, 0.10);
                    }
                    if (arrowV.visible) {
                        arrowV.position.copy(origin);
                        arrowV.setDirection(new THREE.Vector3(dirX * state.vx, state.vy, dirZ * state.vx).normalize());
                        arrowV.setLength(Math.max(0.32, speed * vecScale), 0.22, 0.12);
                    }
                }

                update3DTrail();
            }

            function stepJump(dt) {
                state.jumpTimer += dt;
                sync3DObjects();

                if (state.jumpTimer >= state.jumpDuration) {
                    // 점프 정점(타점 h0) 도달 시 정확히 캐릭터 발사 위치 (playerPos.x, h0, playerPos.z)에서 골대를 향해 발사!
                    state.status = 'running';
                    state.t = 0;
                    state.s = 0;
                    state.x = playerPos.x;
                    state.y = params.releaseHeight;
                    state.z = playerPos.z;
                    state.trail = [{ x: state.x, y: state.y, z: state.z }];
                    btnPause.disabled = false;
                    sync3DObjects();
                    updateDashboard();
                }
            }

            function stepPhysics(dt) {
                const prevS = state.s;
                const prevY = state.y;
                const prevVy = state.vy;

                state.t += dt;
                state.s += state.vx * dt;
                state.vy -= GRAVITY * dt;
                state.y += state.vy * dt;
                state.spinAngle -= (state.vx * dt) / BALL_RADIUS;

                const curXZ = getWorldXZFromProgress(state.s);
                state.x = curXZ.x;
                state.z = curXZ.z;

                // 최고점 포착 (vy가 양수 -> 음수로 바뀌는 순간)
                if (!state.hasCollided && prevVy > 0 && state.vy <= 0 && !state.apexPoint) {
                    state.apexPoint = { x: state.x, y: state.y, z: state.z };
                    update3DEnvironmentAndGuides();
                }

                // 궤적 포인트 기록 (일정 거리 이상 이동했을 때 추가하여 부드러운 리본 유지)
                const lastPt = state.trail[state.trail.length - 1];
                if (!lastPt || Math.hypot(state.x - lastPt.x, state.y - lastPt.y, (state.z - (lastPt.z || 0))) >= 0.04) {
                    state.trail.push({ x: state.x, y: state.y, z: state.z });
                }

                // 1) 득점 판정 (위에서 아래로 림 평면 y = HOOP_HEIGHT 통과 시, 골대 중심까지 진행 거리 s == hoopDistance 기준)
                if (!state.scored && prevY >= HOOP_HEIGHT && state.y < HOOP_HEIGHT) {
                    const ratio = (prevY - HOOP_HEIGHT) / (prevY - state.y);
                    const crossS = prevS + (state.s - prevS) * ratio;
                    const distFromCenter = Math.abs(crossS - params.hoopDistance);
                    if (distFromCenter <= RIM_RADIUS - BALL_RADIUS * 0.45) {
                        state.scored = true;
                        state.celebrateTimer = 3.6; // 3.6초 동안 물만난 광어 댄스 발동!
                        showStatusBanner('🐟 골인!! 물만난 광어 댄스 세리머니! 🎉', 'bg-emerald-500/90 text-white border border-emerald-300');
                    }
                }

                // 2) 백보드 충돌 판정 (3D 월드 X 좌표 HOOP_WORLD_X + backboardX 기준)
                const boardFrontX = HOOP_WORLD_X + backboardX - 0.03;
                const boardBottomY = HOOP_HEIGHT - 0.175;
                const boardTopY = HOOP_HEIGHT + 0.875;
                if (state.vx > 0 && state.x + BALL_RADIUS >= boardFrontX && Math.abs(state.z) <= 0.90) {
                    if (state.y >= boardBottomY && state.y <= boardTopY) {
                        state.x = boardFrontX - BALL_RADIUS;
                        state.s = Math.hypot(state.x - playerPos.x, state.z - playerPos.z);
                        state.vx = -state.vx * RESTITUTION;
                        state.hasCollided = true;
                    }
                }

                // 3) 림 앞/뒤 모서리 충돌 판정 (슛 평면 상 골대 앞/뒤 림 단면 기준)
                const rimPoints = [
                    { rs: params.hoopDistance - RIM_RADIUS, ry: HOOP_HEIGHT },
                    { rs: params.hoopDistance + RIM_RADIUS, ry: HOOP_HEIGHT }
                ];
                for (const rp of rimPoints) {
                    const ds = state.s - rp.rs;
                    const dy = state.y - rp.ry;
                    const dist = Math.hypot(ds, dy);
                    const minDist = BALL_RADIUS + 0.018;
                    if (dist < minDist && dist > 0.0001) {
                        const ns = ds / dist;
                        const ny = dy / dist;
                        state.s = rp.rs + ns * minDist;
                        state.y = rp.ry + ny * minDist;
                        const updatedXZ = getWorldXZFromProgress(state.s);
                        state.x = updatedXZ.x;
                        state.z = updatedXZ.z;

                        const vn = state.vx * ns + state.vy * ny;
                        if (vn < 0) {
                            state.vx -= (1 + RESTITUTION) * vn * ns;
                            state.vy -= (1 + RESTITUTION) * vn * ny;
                            state.hasCollided = true;
                        }
                    }
                }

                // 4) 코트 바닥 충돌 판정
                if (state.y <= BALL_RADIUS) {
                    state.y = BALL_RADIUS;
                    if (Math.abs(state.vy) > 0.65) {
                        state.vy = -state.vy * RESTITUTION;
                        state.vx *= 0.85;
                        state.hasCollided = true;
                    } else {
                        state.vy = 0;
                        state.vx = 0;
                        state.status = 'finished';
                        btnPause.disabled = true;
                        btnStep.disabled = true;
                        if (!state.scored) {
                            showStatusBanner('🏀 슛 종료 · 각도나 속력을 조절해 다시 도전해보세요!', 'bg-slate-800/95 text-slate-200 border border-slate-600');
                        }
                    }
                }

                // 에너지 히스토리 기록
                const vSq = state.vx * state.vx + state.vy * state.vy;
                const ek = 0.5 * MASS * vSq;
                const ep = MASS * GRAVITY * Math.max(0, state.y);
                state.energyHistory.push({ t: state.t, ek, ep, etot: ek + ep });

                updateDashboard();
                sync3DObjects();
            }

            function drawEnergyChart() {
                const rect = chartCanvas.parentElement.getBoundingClientRect();
                const dpr = window.devicePixelRatio || 1;
                if (chartCanvas.width !== Math.floor(rect.width * dpr) || chartCanvas.height !== Math.floor(rect.height * dpr)) {
                    chartCanvas.width = Math.floor(rect.width * dpr);
                    chartCanvas.height = Math.floor(rect.height * dpr);
                }
                const w = chartCanvas.width;
                const h = chartCanvas.height;
                chartCtx.clearRect(0, 0, w, h);

                const padL = 34 * dpr;
                const padR = 10 * dpr;
                const padT = 10 * dpr;
                const padB = 18 * dpr;
                const plotW = w - padL - padR;
                const plotH = h - padT - padB;

                // 배경 격자
                chartCtx.strokeStyle = 'rgba(148, 163, 184, 0.15)';
                chartCtx.lineWidth = 1 * dpr;
                for (let i = 0; i <= 4; i++) {
                    const gy = padT + (plotH * i) / 4;
                    chartCtx.beginPath();
                    chartCtx.moveTo(padL, gy);
                    chartCtx.lineTo(w - padR, gy);
                    chartCtx.stroke();
                }

                const hist = state.energyHistory;
                const maxE = Math.max(35, ...hist.map(d => d.etot * 1.15));
                const maxT = Math.max(1.8, state.t * 1.05);

                // Y축 눈금 텍스트
                chartCtx.fillStyle = '#94a3b8';
                chartCtx.font = `${10 * dpr}px monospace`;
                chartCtx.textAlign = 'right';
                chartCtx.fillText(`${Math.round(maxE)}J`, padL - 4 * dpr, padT + 8 * dpr);
                chartCtx.fillText(`${Math.round(maxE / 2)}J`, padL - 4 * dpr, padT + plotH / 2 + 3 * dpr);
                chartCtx.fillText('0J', padL - 4 * dpr, padT + plotH);

                if (hist.length < 2) return;

                function plotCurve(key, color) {
                    chartCtx.strokeStyle = color;
                    chartCtx.lineWidth = 2 * dpr;
                    chartCtx.beginPath();
                    for (let i = 0; i < hist.length; i++) {
                        const px = padL + (hist[i].t / maxT) * plotW;
                        const py = padT + plotH - (hist[i][key] / maxE) * plotH;
                        if (i === 0) chartCtx.moveTo(px, py);
                        else chartCtx.lineTo(px, py);
                    }
                    chartCtx.stroke();
                }

                plotCurve('etot', '#34d399'); // 역학적 에너지 E (초록)
                plotCurve('ek', '#fb7185');   // 운동 에너지 Ek (빨강)
                plotCurve('ep', '#38bdf8');   // 퍼텐셜 에너지 Ep (파랑)
            }

            // -------------------------------------------------------------------------
            // 8. 시뮬레이션 제어 (초기화 / 발사 / 일시정지 / 프레임 탐색)
            // -------------------------------------------------------------------------
            function resetSimulation() {
                const rad = (params.angleDeg * Math.PI) / 180;
                const vx0 = params.speed * Math.cos(rad);
                const vy0 = params.speed * Math.sin(rad);

                state.status = 'ready';
                state.t = 0;
                state.jumpTimer = 0;
                state.celebrateTimer = 0;
                state.s = 0;
                state.x = playerPos.x;
                state.y = params.releaseHeight;
                state.z = playerPos.z;
                state.vx = vx0;
                state.vy = vy0;
                state.spinAngle = 0;
                state.trail = [];
                state.apexPoint = null;
                state.scored = false;
                state.hasCollided = false;

                const ek0 = 0.5 * MASS * params.speed * params.speed;
                const ep0 = MASS * GRAVITY * params.releaseHeight;
                state.energyHistory = [{ t: 0, ek: ek0, ep: ep0, etot: ek0 + ep0 }];

                btnPause.disabled = true;
                btnStep.disabled = true;
                pauseText.textContent = '일시정지';
                hideStatusBanner();

                updateDistanceZoneUI();
                update3DEnvironmentAndGuides();
                sync3DObjects();
                updateDashboard();
            }

            function launchSimulation() {
                if (state.status !== 'ready') {
                    resetSimulation();
                }
                state.status = 'jumping';
                state.jumpTimer = 0;
                // 타점이 높을수록 점프 체공 시간을 약간 더 길게 부여해 자연스러운 도약 연출
                state.jumpDuration = params.releaseHeight > 1.58
                    ? Math.min(0.48, 0.28 + (params.releaseHeight - 1.58) * 0.12)
                    : 0.24;
                btnPause.disabled = true;
                btnStep.disabled = true;
                pauseText.textContent = '일시정지';
                hideStatusBanner();
            }

            // -------------------------------------------------------------------------
            // 9. 이벤트 리스너 바인딩 (슬라이더, 프리셋, 3D 카메라 오빗, 조이스틱, 애니메이션 루프)
            // -------------------------------------------------------------------------
            function setPlayerDistanceAlongCurrentRay(newDist) {
                const clamped = Math.min(15.0, Math.max(1.0, newDist));
                // 현재 골대를 향하는 좌우 각도(aimYaw)를 유지하면서 거리만 조절
                playerPos.x = HOOP_WORLD_X - Math.cos(playerPos.aimYaw) * clamped;
                playerPos.z = -Math.sin(playerPos.aimYaw) * clamped;
                updatePlayerDistanceAndAim(true);
            }

            function syncParameter(key, val, min, max, sliderEl, numEl) {
                const clamped = Math.min(max, Math.max(min, val));
                params[key] = clamped;
                sliderEl.value = clamped;
                numEl.value = key === 'distance' || key === 'releaseHeight' || key === 'hoopDistance'
                    ? clamped.toFixed(2)
                    : clamped.toFixed(1);
                if (key === 'hoopDistance') {
                    setPlayerDistanceAlongCurrentRay(clamped);
                }
                resetSimulation();
            }

            inputAngle.addEventListener('input', e => syncParameter('angleDeg', parseFloat(e.target.value), 15, 85, inputAngle, numAngle));
            numAngle.addEventListener('change', e => syncParameter('angleDeg', parseFloat(e.target.value) || 52, 15, 85, inputAngle, numAngle));

            inputSpeed.addEventListener('input', e => syncParameter('speed', parseFloat(e.target.value), 0, 16, inputSpeed, numSpeed));
            numSpeed.addEventListener('change', e => syncParameter('speed', parseFloat(e.target.value) || 0, 0, 16, inputSpeed, numSpeed));

            inputHeight.addEventListener('input', e => syncParameter('releaseHeight', parseFloat(e.target.value), 0.2, 3.5, inputHeight, numHeight));
            numHeight.addEventListener('change', e => syncParameter('releaseHeight', parseFloat(e.target.value) || 2.1, 0.2, 3.5, inputHeight, numHeight));

            inputDistance.addEventListener('input', e => {
                syncParameter('hoopDistance', parseFloat(e.target.value), 1.0, 15.0, inputDistance, numDistance);
                setCameraPreset(camControl.mode);
            });
            numDistance.addEventListener('change', e => {
                syncParameter('hoopDistance', parseFloat(e.target.value) || 4.6, 1.0, 15.0, inputDistance, numDistance);
                setCameraPreset(camControl.mode);
            });

            document.getElementById('btnDistMinus').addEventListener('click', () => {
                syncParameter('hoopDistance', +(params.hoopDistance - 0.1).toFixed(2), 1.0, 15.0, inputDistance, numDistance);
                setCameraPreset(camControl.mode);
            });
            document.getElementById('btnDistPlus').addEventListener('click', () => {
                syncParameter('hoopDistance', +(params.hoopDistance + 0.1).toFixed(2), 1.0, 15.0, inputDistance, numDistance);
                setCameraPreset(camControl.mode);
            });

            // 원클릭 코트 거리 프리셋 버튼
            document.querySelectorAll('.preset-dist-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    const d = parseFloat(btn.dataset.dist);
                    setPlayerDistanceAlongCurrentRay(d);

                    if (toggleAutoAimPreset.checked) {
                        const ang = parseFloat(btn.dataset.angle);
                        const spd = parseFloat(btn.dataset.speed);
                        params.angleDeg = ang;
                        params.speed = spd;
                        inputAngle.value = ang;
                        numAngle.value = ang.toFixed(1);
                        inputSpeed.value = spd;
                        numSpeed.value = spd.toFixed(1);
                    }
                    setCameraPreset(camControl.mode);
                    resetSimulation();
                });
            });

            // 시각화 토글 체크박스
            [toggleTrail, toggleVectors, toggleAimGuide, toggleApexMarker].forEach(el => {
                el.addEventListener('change', () => {
                    update3DEnvironmentAndGuides();
                    sync3DObjects();
                });
            });

            // 카메라 시점 프리셋 버튼
            document.querySelectorAll('.cam-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.cam-btn').forEach(b => {
                        b.classList.remove('bg-sky-600', 'text-white', 'font-semibold');
                        b.classList.add('text-slate-300');
                    });
                    btn.classList.add('bg-sky-600', 'text-white', 'font-semibold');
                    btn.classList.remove('text-slate-300');
                    setCameraPreset(btn.dataset.cam);
                });
            });

            // 배속 버튼
            document.querySelectorAll('.speed-btn').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.speed-btn').forEach(b => {
                        b.classList.remove('bg-orange-500', 'text-white', 'font-semibold');
                        b.classList.add('text-slate-300');
                    });
                    btn.classList.add('bg-orange-500', 'text-white', 'font-semibold');
                    btn.classList.remove('text-slate-300');
                    state.timeScale = parseFloat(btn.dataset.speed);
                });
            });

            // 메인 액션 버튼
            btnLaunch.addEventListener('click', launchSimulation);

            btnPause.addEventListener('click', () => {
                if (state.status === 'running') {
                    state.status = 'paused';
                    pauseText.textContent = '계속 재생';
                    btnStep.disabled = false;
                } else if (state.status === 'paused') {
                    state.status = 'running';
                    pauseText.textContent = '일시정지';
                    btnStep.disabled = true;
                }
            });

            btnStep.addEventListener('click', () => {
                if (state.status === 'paused') {
                    stepPhysics(0.02);
                }
            });

            btnReset.addEventListener('click', resetSimulation);

            // -------------------------------------------------------------------------
            // ★ 가상 아날로그 조이스틱 & 키보드(WASD/방향키) 캐릭터 코트 이동 컨트롤러
            // -------------------------------------------------------------------------
            const joystickBase = document.getElementById('joystickBase');
            const joystickKnob = document.getElementById('joystickKnob');
            let joyState = {
                active: false,
                pointerId: null,
                vecX: 0, // -1 ~ +1 (좌우 입력)
                vecY: 0  // -1 ~ +1 (상하 입력: 위쪽이 골대 방향 전진)
            };
            const keysPressed = {};

            function updateJoystickVisual(nx, ny) {
                const maxOffset = 32;
                joystickKnob.style.transform = `translate(${nx * maxOffset}px, ${ny * maxOffset}px)`;
            }

            function handleJoystickMove(clientX, clientY) {
                const rect = joystickBase.getBoundingClientRect();
                const centerX = rect.left + rect.width / 2;
                const centerY = rect.top + rect.height / 2;
                const maxRadius = rect.width * 0.42;

                let dx = clientX - centerX;
                let dy = clientY - centerY;
                const dist = Math.hypot(dx, dy);
                if (dist > maxRadius) {
                    dx = (dx / dist) * maxRadius;
                    dy = (dy / dist) * maxRadius;
                }
                joyState.vecX = dx / maxRadius;
                joyState.vecY = dy / maxRadius;
                updateJoystickVisual(joyState.vecX, joyState.vecY);
            }

            joystickBase.addEventListener('pointerdown', e => {
                e.stopPropagation();
                joyState.active = true;
                joyState.pointerId = e.pointerId;
                joystickBase.setPointerCapture(e.pointerId);
                handleJoystickMove(e.clientX, e.clientY);
            });

            joystickBase.addEventListener('pointermove', e => {
                if (!joyState.active || e.pointerId !== joyState.pointerId) return;
                e.stopPropagation();
                handleJoystickMove(e.clientX, e.clientY);
            });

            function resetJoystick(e) {
                if (e && joyState.pointerId !== null && e.pointerId !== joyState.pointerId) return;
                joyState.active = false;
                joyState.pointerId = null;
                joyState.vecX = 0;
                joyState.vecY = 0;
                updateJoystickVisual(0, 0);
            }
            joystickBase.addEventListener('pointerup', resetJoystick);
            joystickBase.addEventListener('pointercancel', resetJoystick);

            window.addEventListener('keydown', e => {
                if (['INPUT', 'TEXTAREA'].includes(document.activeElement?.tagName)) return;
                keysPressed[e.code] = true;
            });
            window.addEventListener('keyup', e => {
                keysPressed[e.code] = false;
            });

            function updatePlayerMovement(dt) {
                let inX = joyState.vecX;
                let inY = joyState.vecY;

                // 키보드 입력 합산 (W/↑: 골대 쪽 전진, S/↓: 후진, A/←: 왼쪽, D/→: 오른쪽)
                if (keysPressed['KeyW'] || keysPressed['ArrowUp']) inY -= 1;
                if (keysPressed['KeyS'] || keysPressed['ArrowDown']) inY += 1;
                if (keysPressed['KeyA'] || keysPressed['ArrowLeft']) inX -= 1;
                if (keysPressed['KeyD'] || keysPressed['ArrowRight']) inX += 1;

                const mag = Math.hypot(inX, inY);
                if (mag < 0.08) {
                    if (playerPos.isMoving) {
                        playerPos.isMoving = false;
                        updateCharacterPose();
                    }
                    return;
                }

                // 이동 입력 시 세리머니 중이었다면 즉시 달리기 자세로 전환
                state.celebrateTimer = 0;

                // 입력 정규화
                const normX = mag > 1 ? inX / mag : inX;
                const normY = mag > 1 ? inY / mag : inY;

                // 현재 카메라 시점 기준으로 코트 위 전후좌우 이동 벡터 변환
                const forwardX = -Math.sin(camControl.yaw);
                const forwardZ = -Math.cos(camControl.yaw);
                const rightX = Math.cos(camControl.yaw);
                const rightZ = -Math.sin(camControl.yaw);

                const moveSpeed = 4.6; // 초당 4.6m 달리기 속도
                const worldVx = (forwardX * (-normY) + rightX * normX) * moveSpeed;
                const worldVz = (forwardZ * (-normY) + rightZ * normX) * moveSpeed;

                let nextX = playerPos.x + worldVx * dt;
                let nextZ = playerPos.z + worldVz * dt;

                // 코트 경계 제한 (골대 뒤쪽 및 사이드라인/하프라인 안쪽 유지)
                nextX = Math.max(HOOP_WORLD_X - 14.5, Math.min(HOOP_WORLD_X - 0.8, nextX));
                nextZ = Math.max(-7.2, Math.min(7.2, nextZ));

                // 골대 림 직하방 최소 거리(1.0m) 보장
                const dxHoop = HOOP_WORLD_X - nextX;
                const dzHoop = -nextZ;
                const distHoop = Math.hypot(dxHoop, dzHoop);
                if (distHoop < 1.0 && distHoop > 0.001) {
                    nextX = HOOP_WORLD_X - (dxHoop / distHoop) * 1.0;
                    nextZ = (dzHoop / distHoop) * 1.0;
                }

                playerPos.x = nextX;
                playerPos.z = nextZ;
                playerPos.isMoving = true;
                playerPos.runCycle += dt * 15;

                // 이동한 위치에서 골대 중심까지의 거리와 좌우 조준각(aimYaw) 자동 업데이트
                updatePlayerDistanceAndAim(true);

                // 만약 이전 슛이 종료된 상태라면 즉시 새로운 위치에서 준비 상태로 전환
                if (state.status === 'finished') {
                    resetSimulation();
                } else if (state.status === 'ready') {
                    updateDistanceZoneUI();
                    update3DEnvironmentAndGuides();
                    sync3DObjects();
                }
            }

            // 3D 마우스/터치 오빗 회전 및 휠 줌
            simCanvas.addEventListener('mousedown', e => {
                camControl.isDragging = true;
                camControl.lastX = e.clientX;
                camControl.lastY = e.clientY;
            });
            window.addEventListener('mousemove', e => {
                if (!camControl.isDragging) return;
                const dx = e.clientX - camControl.lastX;
                const dy = e.clientY - camControl.lastY;
                camControl.lastX = e.clientX;
                camControl.lastY = e.clientY;

                camControl.yaw -= dx * 0.007;
                camControl.pitch = Math.max(0.03, Math.min(1.25, camControl.pitch + dy * 0.006));
                updateCameraPosition();
            });
            window.addEventListener('mouseup', () => {
                camControl.isDragging = false;
            });
            simCanvas.addEventListener('wheel', e => {
                e.preventDefault();
                camControl.distance = Math.max(3.5, Math.min(28.0, camControl.distance + e.deltaY * 0.008));
                updateCameraPosition();
            }, { passive: false });

            // 모바일 터치 시점 회전
            simCanvas.addEventListener('touchstart', e => {
                if (e.touches.length === 1) {
                    camControl.isDragging = true;
                    camControl.lastX = e.touches[0].clientX;
                    camControl.lastY = e.touches[0].clientY;
                }
            }, { passive: true });
            simCanvas.addEventListener('touchmove', e => {
                if (!camControl.isDragging || e.touches.length !== 1) return;
                const dx = e.touches[0].clientX - camControl.lastX;
                const dy = e.touches[0].clientY - camControl.lastY;
                camControl.lastX = e.touches[0].clientX;
                camControl.lastY = e.touches[0].clientY;

                camControl.yaw -= dx * 0.008;
                camControl.pitch = Math.max(0.03, Math.min(1.25, camControl.pitch + dy * 0.007));
                updateCameraPosition();
            }, { passive: true });
            simCanvas.addEventListener('touchend', () => {
                camControl.isDragging = false;
            });

            // 반응형 3D 렌더러 크기 조정
            function handleResize() {
                const rect = canvasWrapper.getBoundingClientRect();
                const width = Math.max(300, rect.width);
                const height = Math.max(360, rect.height);
                camera.aspect = width / height;
                camera.updateProjectionMatrix();
                renderer.setSize(width, height, false);
                drawEnergyChart();
            }
            window.addEventListener('resize', handleResize);

            // 메인 애니메이션 루프
            let lastTimestamp = null;
            function animateLoop(timestamp) {
                if (lastTimestamp === null) lastTimestamp = timestamp;
                const rawDt = Math.min(0.05, (timestamp - lastTimestamp) / 1000);
                lastTimestamp = timestamp;

                // 조이스틱 / 키보드 캐릭터 이동 처리
                updatePlayerMovement(rawDt);

                // 물만난 광어 댄스 타이머 진행
                if (state.celebrateTimer > 0) {
                    state.celebrateTimer = Math.max(0, state.celebrateTimer - rawDt);
                    updateCharacterPose();
                }

                if (state.status === 'jumping') {
                    stepJump(rawDt * Math.max(0.5, state.timeScale));
                } else if (state.status === 'running') {
                    const scaledDt = rawDt * state.timeScale;
                    const subSteps = 4;
                    const subDt = scaledDt / subSteps;
                    for (let i = 0; i < subSteps; i++) {
                        if (state.status !== 'running') break;
                        stepPhysics(subDt);
                    }
                }

                // 최고점 마커 부드러운 회전
                if (apexMarkerGroup.visible) {
                    apexDiamond.rotation.y += 0.03;
                }

                renderer.render(scene, camera);
                requestAnimationFrame(animateLoop);
            }

            // 초기 실행
            window.onload = function () {
                handleResize();
                updatePlayerDistanceAndAim(true);
                setCameraPreset('quarter');
                resetSimulation();
                requestAnimationFrame(animateLoop);
            };
        })();
    </script>
</body>
</html>
